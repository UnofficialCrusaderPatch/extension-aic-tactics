cmake_minimum_required(VERSION 3.15)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(aic_tactics_policy LANGUAGES CXX)

add_library(aic_tactics_policy STATIC src/recruitment.cpp src/random.cpp src/composition.cpp src/targets.cpp src/incidents.cpp)
target_include_directories(aic_tactics_policy PUBLIC include)
set_target_properties(aic_tactics_policy PROPERTIES CXX_STANDARD 98 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)

enable_testing()
add_executable(composition_tests tests/composition_tests.cpp)
target_link_libraries(composition_tests PRIVATE aic_tactics_policy)
set_target_properties(composition_tests PROPERTIES CXX_STANDARD 98 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
  target_compile_options(composition_tests PRIVATE /UNDEBUG)
else()
  target_compile_options(composition_tests PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
endif()
add_test(NAME defense_composition COMMAND composition_tests)
add_executable(recruitment_tests tests/recruitment_tests.cpp)
target_link_libraries(recruitment_tests PRIVATE aic_tactics_policy)
set_target_properties(recruitment_tests PROPERTIES CXX_STANDARD 98 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
  target_compile_options(recruitment_tests PRIVATE /UNDEBUG)
else()
  target_compile_options(aic_tactics_policy PRIVATE -Wall -Wextra -Werror -pedantic)
  target_compile_options(recruitment_tests PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
endif()
add_test(NAME recruitment_policy COMMAND recruitment_tests)

add_executable(random_tests tests/random_tests.cpp)
target_link_libraries(random_tests PRIVATE aic_tactics_policy)
set_target_properties(random_tests PROPERTIES CXX_STANDARD 98 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
  target_compile_options(random_tests PRIVATE /UNDEBUG)
else()
  target_compile_options(random_tests PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
endif()
add_test(NAME bounded_random_draw COMMAND random_tests)

foreach(policy_test target incident)
  add_executable(${policy_test}_tests tests/${policy_test}_tests.cpp)
  target_link_libraries(${policy_test}_tests PRIVATE aic_tactics_policy)
  set_target_properties(${policy_test}_tests PROPERTIES CXX_STANDARD 98 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  if(MSVC)
    target_compile_options(${policy_test}_tests PRIVATE /W4 /WX /UNDEBUG)
  else()
    target_compile_options(${policy_test}_tests PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
  endif()
  add_test(NAME ${policy_test}_policy COMMAND ${policy_test}_tests)
endforeach()

  target_include_directories(aic_tactics_shc141 PUBLIC include)
  target_compile_options(aic_tactics_shc141 PRIVATE /W4 /WX)
  add_library(aicTactics SHARED src/runtime.cpp src/combat.cpp src/army.cpp src/raids.cpp src/integrity.cpp src/lua_module.cpp)
  target_link_libraries(aicTactics PRIVATE aic_tactics_policy aic_tactics_shc141 kernel32)
  target_compile_options(aicTactics PRIVATE /W4 /WX)

    " /MANIFEST:NO /BASE:0x400000 /FIXED /MERGE:.text=.ztext /MERGE:.rdata=.zrdata /MERGE:.data=.zdata")
  if(MSVC_VERSION GREATER_EQUAL 1500)
    set_property(TARGET shc141_probe_tests APPEND_STRING PROPERTY LINK_FLAGS " /DYNAMICBASE:NO")
  endif()
  # Requires a private licensed reference image and runs through its hash-checking
  # Python wrapper. It is deliberately not a public CI gameplay test.
endif()
